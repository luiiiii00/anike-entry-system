type SpecExtra = Partial<Omit<Question, "id" | "label">>;

/** Pregunta SCORE de la especificación (escala genérica pendiente de fuente). */

  id,
  label,
  kind: "SCORE",
  options: SPEC_SCALE,
  pendingScale: true,
  ...extra,
});
const sv = (id: string, label: string, extra: SpecExtra = {}): Question =>
  sq(id, label, { kind: "SCORE_VALIDATION", ...extra });
const sav = (id: string, label: string, extra: SpecExtra = {}): Question =>
  sq(id, label, {
    kind: "SCORE_AUTO_VALIDATION",

    ...extra,
  });

  id,
  label,
  kind: "VALIDATION",
  validationOnly: true,
  options: VALIDATION_YN,
  ...extra,
});

/**
 * Preguntas de VALIDACIÓN del nuevo cuestionario que no se cumplen. Nunca son
 * HARD: dejan la evaluación CONDICIONAL. VALIDATION pura falla con "no";
 * SCORE + VALIDATION falla con factor 0.
 */
export function failedValidations(
  answers: Record<string, string>,
  activeIds?: Set<string>,
): string[] {
  const failed: string[] = [];
  for (const section of SECTIONS) {
    for (const group of section.groups) {
      for (const q of group.questions) {
        const k = q.kind;
        if (
          k !== "VALIDATION" &&
          k !== "CONDITIONAL_VALIDATION" &&
          k !== "SCORE_VALIDATION" &&
          k !== "SCORE_AUTO_VALIDATION"
        )
          continue;
        if (activeIds && !activeIds.has(q.id)) continue;
        const value = answers[q.id];
        if (value === undefined || value === "") continue;
        const opt = q.options.find((o) => o.v === value);
        if (opt && opt.pts === 0) failed.push(q.id);
      }
    }
  }
  return failed;
}

/**
 * Limpia respuestas de preguntas condicionales que dejaron de estar activas al
 * cambiar la metadata que las condiciona (MOMENTUM ↔ THREE_BODY, GIRO ↔ LÍMITE).
 */
export function pruneInactiveAnswers(
  answers: Record<string, string>,
  setupId: string | null | undefined,
): Record<string, string> {
  const questions = getActiveQuestionsBySetup(setupId);
  if (questions.length === 0) return answers;
  const next = { ...answers };
  for (const q of questions) {
    if (!q.condition) continue;
    const declared = next[q.condition.questionId];
    const matches =
      declared !== undefined &&
      declared !== "" &&
      (q.condition.value === ANY_CONDITION_VALUE || declared === q.condition.value);
    if (!matches) delete next[q.id];
  }
  return next;
}
